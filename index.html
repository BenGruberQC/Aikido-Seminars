/** * Normalise les tarifs (colonnes P, Q, R) en ajoutant explicitement "CAD" ou "USD" * en se basant sur la province/état (colonne L) et l'adresse civique (colonne M). */ function normaliserDevisesTarifs() { const ss = SpreadsheetApp.getActiveSpreadsheet(); // Remplacez 'Séminaires' par le nom exact de votre onglet principal si différent const sheet = ss.getSheetByName('Séminaires') || ss.getActiveSheet(); const lastRow = sheet.getLastRow(); if (lastRow < 2) return; // Seulement l'en-tête ou vide // Lecture des colonnes L à R (L: Province, M: Adresse, N: Lat, O: Lng, P: Tarif, Q: Tarif J1, R: Tarif J2) // Col L = 12, Nb colonnes = 7 (L, M, N, O, P, Q, R) const range = sheet.getRange(2, 12, lastRow - 1, 7); const data = range.getValues(); const US_STATES = new Set([ 'AL', 'AK', 'AZ', 'AR', 'CA', 'CO', 'CT', 'DE', 'FL', 'GA', 'HI', 'ID', 'IL', 'IN', 'IA', 'KS', 'KY', 'LA', 'ME', 'MD', 'MA', 'MI', 'MN', 'MS', 'MO', 'MT', 'NE', 'NV', 'NH', 'NJ', 'NM', 'NY', 'NC', 'ND', 'OH', 'OK', 'OR', 'PA', 'RI', 'SC', 'SD', 'TN', 'TX', 'UT', 'VT', 'VA', 'WA', 'WV', 'WI', 'WY' ]); const CA_PROVINCES = new Set([ 'AB', 'BC', 'MB', 'NB', 'NL', 'NS', 'NT', 'NU', 'ON', 'PE', 'QC', 'SK', 'YT' ]); let modifications = false; for (let i = 0; i < data.length; i++) { const province = String(data[i][0] || '').trim().toUpperCase(); const adresse = String(data[i][1] || '').trim().toUpperCase(); // Détection pays const isCanada = CA_PROVINCES.has(province) || adresse.includes('CANADA') || /\b[A-Z]\d[A-Z]\s?\d[A-Z]\d\b/.test(adresse); const isUSA = US_STATES.has(province) || adresse.includes('USA') || adresse.includes('UNITED STATES') || /\b\d{5}(-\d{4})?\b/.test(adresse); let devise = ''; if (isCanada) devise = 'CAD'; else if (isUSA) devise = 'USD'; if (!devise) continue; // Colonnes P (index 4), Q (index 5), R (index 6) dans notre extrait for (let colIdx = 4; colIdx <= 6; colIdx++) { let val = String(data[i][colIdx] || '').trim(); if (!val) continue; // Si la devise est déjà présente, on ne touche pas if (val.toUpperCase().includes('CAD') || val.toUpperCase().includes('USD')) { continue; } // Si c'est juste "$120" -> "120 $ CAD" // Si c'est "120" -> "120 $ CAD" // Si c'est "$120 ($90 Membres)" -> "$120 ($90 Membres) CAD" if (/^\$?\s*\d+(\.\d+)?\s*\$?$/.test(val)) { const montant = val.replace(/\$/g, '').trim(); data[i][colIdx] = `${montant} $ ${devise}`; modifications = true; } else { data[i][colIdx] = `${val} ${devise}`; modifications = true; } } } if (modifications) { range.setValues(data); } }
